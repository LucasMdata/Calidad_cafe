CREATE PROCEDURE SP_CARGA_DIM_FACT

AS
BEGIN

INSERT INTO DIM_Pais (Pais)
SELECT DISTINCT C.Pais
FROM INT_CAFE C
LEFT JOIN DIM_Pais P ON C.Pais = P.Pais
WHERE P.Id_Pais IS NULL;

INSERT INTO DIM_Variedad (Variedad)
SELECT DISTINCT C.Variedad
FROM INT_CAFE C
LEFT JOIN DIM_Variedad V ON C.Variedad = V.Variedad
WHERE V.Id_variedad IS NULL;

INSERT INTO DIM_REGION(Region)
SELECT DISTINCT C.Region
FROM INT_CAFE C
LEFT JOIN DIM_REGION R ON C.Region = R.Region
WHERE R.Id_Region IS NULL;

INSERT INTO DIM_METODO_PROCESADO (Metodo_procesado)
SELECT DISTINCT C.Metodo_procesado
FROM INT_CAFE C
LEFT JOIN DIM_METODO_PROCESADO MP ON C.metodo_procesado = MP.Metodo_procesado
WHERE MP.Id_Metodo_procesado IS NULL;

INSERT INTO FACT_ANALISIS (P.Id_Pais,
	Id_Region,
	Id_Variedad,
	Id_Metodo_procesado,
	Aroma,
	Sabor,
	Regusto,
	Acidez,
	Cuerpo,
	Balance,
	Uniformidad,
	Taza_limpia,
	Dulzor,
	General,
	Defecto,
	Puntos_totales,
	Porcentaje_humedad,
	Color)
SELECT

	P.Id_Pais,
	R.Id_Region,
	V.Id_Variedad,
	MP.Id_Metodo_procesado,
	C.Aroma,
	C.Sabor,
	C.Regusto,
	C.Acidez,
	C.Cuerpo,
	C.Balance,
	C.Uniformidad,
	C.Taza_limpia,
	C.Dulzor,
	C.General,
	C.Defecto,
	C.Puntos_totales,
	C.Porcentaje_humedad,
	C.Color
FROM INT_CAFE C
JOIN DIM_PAIS P ON C.Pais = P.Pais
JOIN DIM_REGION R ON R.Region = C.Region
JOIN DIM_VARIEDAD V ON V.Variedad = C.Variedad
JOIN DIM_METODO_PROCESADO MP ON MP.Metodo_procesado = C.Metodo_procesado
LEFT JOIN FACT_ANALISIS F ON
	
	F.Id_Region = R.Id_Region AND
	F.Id_Variedad = V.Id_variedad AND
	F.Id_Metodo_procesado = MP.Id_Metodo_procesado and
	F.Aroma = c.Aroma and
	F.Sabor = c.Sabor and
	F.Regusto = c.Regusto and
	F.Acidez = c.Acidez and
	F.Cuerpo = c.Cuerpo and
	F.Balance = c.Balance and
	F.Uniformidad = c.Uniformidad and
	F.Taza_limpia = c.Taza_limpia and
	F.Dulzor = c.Dulzor and
	F.General = c.General and
	F.Defecto = c.Defecto and
	F.Puntos_totales = c.Puntos_totales and
	F.Porcentaje_humedad = c.Porcentaje_humedad and
	F.Color = c.Color
	WHERE F.Id_analisis IS NULL	   
END
